import json

def extract_reviews_to_json(input_data, output_file='extracted_reviews.json'):
    """
    Extract all reviews and comments from the input JSON data and save them to a new JSON file.
    :param input_data: The original JSON data representing Reddit conversations.
    :param output_file: The file to which the reviews will be saved (default: 'extracted_reviews.json').
    """
    # List to store all extracted reviews and comments
    extracted_reviews = []

    def gather_comments(comments):
        """
        Recursive function to gather comments and their replies.
        """
        extracted_comments = []
        for comment in comments:
            # Add the comment body
            extracted_comments.append({
                'text': comment['body'],
                'user': comment.get('user', 'anonymous'),  # If no user, assign anonymous
                'time': comment.get('time', 'unknown')  # If no time, set to unknown
            })
            # If there are replies, gather them recursively
            if 'replies' in comment and comment['replies']:
                extracted_comments.extend(gather_comments(comment['replies']))
        return extracted_comments

    for post in input_data:
        # Extract the main post body
        post_data = {
            'post_title': post['post_title'],
            'post_body': post['post_body'],
            'post_user': post.get('post_user', 'anonymous'),
            'post_time': post.get('post_time', 'unknown'),
            'comments': []
        }
        
        # Extract the comments and replies recursively
        post_data['comments'] = gather_comments(post['comments'])

        # Append the extracted post and comments to the output list
        extracted_reviews.append(post_data)

    # Write the extracted reviews to the specified JSON file
    with open(output_file, 'w') as outfile:
        json.dump(extracted_reviews, outfile, indent=4)

    print(f"Reviews and comments successfully extracted and saved to {output_file}")

# Example usage
if __name__ == "__main__":
    # Load the sample JSON data (replace with actual JSON file if available)
    with open('data/costa/2024-10-13-02-06-33.json', 'r') as f:
        data = json.load(f)

    # Extract reviews and comments to a separate JSON file
    extract_reviews_to_json(data, output_file='extracted_reviews.json')
